(function () {
  'use strict';

  /* Le fichier de la page de mentions légales. */
  var PAGE = 'mentions-legales.html';

  /* La colonne « Sections » est la dernière liste de liens du pied de page. */
  var lists = document.querySelectorAll('footer .footer-nav-list');
  var list = lists[lists.length - 1];
  if (!list) return;

  var li = document.createElement('li');
  li.setAttribute('role', 'listitem');
  li.innerHTML =
    '<a class="ml-link" href="' + PAGE + '" target="_blank" rel="noopener">' +

      '<span>Mentions légales</span>' +
      '<i class="fas fa-arrow-up-right-from-square ml-ext" aria-hidden="true"></i>' +
      '<span class="visually-hidden"> (nouvel onglet)</span>' +
    '</a>';
  list.appendChild(li);
  var link = li.querySelector('a');

  function txt(el) { return el ? (el.innerText || el.textContent || '').replace(/\s+/g, ' ').trim() : ''; }

  /* Le nom affiché sur le CV, lu au moment du clic : il suit donc les modifications faites en mode édition ou par import de CV. */
  function readName() {
    var typed  = txt(document.getElementById('typing-name'));
    var strong = txt(document.querySelector('.about-text p:first-of-type strong:first-of-type'));
    var n = (strong && strong.length >= typed.length) ? strong : (typed || strong);
    return /^pr[ée]nom nom$/i.test(n) ? '' : n;
  }
  function readEmail() {
    var a = document.querySelector('footer a[href^="mailto:"]') || document.querySelector('a[href^="mailto:"]');
    return a ? decodeURIComponent(a.getAttribute('href').replace(/^mailto:/i, '').split('?')[0]).trim() : '';
  }
  /* La ville : le début du dernier paragraphe de localisation du pied de page, avant le « · ». */
  function readPlace() {
    var locs = document.querySelectorAll('footer .footer-location');
    var last = locs[locs.length - 1];
    return last ? txt(last).split('·')[0].trim() : '';
  }

  /* On construit l'adresse juste avant l'ouverture. Les informations passent après le « # » :
     elles restent dans le navigateur et ne sont jamais envoyées au serveur. */
  function updateHref() {
    var p = new URLSearchParams();
    var nom = readName(), email = readEmail(), lieu = readPlace(), lang = '', scale = '';
    if (nom)   p.set('nom', nom);
    if (email) p.set('email', email);
    if (lieu)  p.set('lieu', lieu);
    try { lang = localStorage.getItem('portfolio_lang') || ''; } catch (e) {}
    p.set('lang', lang || 'fr');
    p.set('theme', document.documentElement.getAttribute('data-theme') === 'light' ? 'light' : 'dark');
    scale = document.documentElement.style.getPropertyValue('--tz-scale');
    if (scale && parseFloat(scale) !== 1) p.set('scale', String(parseFloat(scale)));
    link.href = PAGE + '#' + p.toString();
  }
  ['mouseenter', 'focus', 'mousedown', 'touchstart', 'click', 'contextmenu'].forEach(function (ev) {
    link.addEventListener(ev, updateHref, { passive: true });
  });
  updateHref();
})();

Ce que fait la page

Fonctionne avec n’importe quel nom : 
le nom est lu sur le CV au moment du clic, donc il suit aussi les changements faits en mode édition ou par import de CV.

Si la page est ouverte directement, sans passer par le lien, elle parle de « la personne titulaire de ce CV portfolio ».

Cette formule s’accorde avec la grammaire de chaque langue (« del titolare », « dem Inhaber »…), et en français le « de » s’élide devant un prénom à voyelle (« d’Anne »).

Les 7 langues du CV : 
les traductions sont écrites dans la page elle-même, sans passer par Google Traduction, ce qui est plus fiable pour un texte juridique. 
Une mention précise que la version française fait foi.

Lisibilité, y compris pour les malvoyants : 
le contraste des textes va de 9,6:1 à 20:1 en clair comme en sombre, au-dessus du niveau AAA (7:1).

On y trouve aussi les boutons A− / A+ jusqu’à 135 %, une navigation complète au clavier avec un contour jaune bien visible, et un lien « Aller au contenu ».

Les lecteurs d’écran annoncent les changements, et la page respecte le réglage « réduire les animations ».
