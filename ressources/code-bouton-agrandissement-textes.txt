<script>
(function () {
  'use strict';

  /* Les réglages : de 100 % à 135 %, par pas de 5 %. Pour autoriser une réduction sous la taille d'origine, passez MIN à 0.9 par exemple. */
  var MIN = 1, MAX = 1.35, STEP = 0.05;
  /* La clé de mémorisation : le choix du visiteur est conservé d'une visite à l'autre. */
  var KEY = 'portfolio_text_scale';
  var html = document.documentElement;
  var scale = 1;

  /* Arrondi à 2 décimales et bornage entre MIN et MAX, pour éviter les 1.1500000002 des calculs en virgule flottante. */
  function clamp(v) { v = Math.round(v * 100) / 100; return Math.min(MAX, Math.max(MIN, v)); }

  /* ── 1. Les boutons, insérés dans le groupe du bouton thème, juste devant lui ── */
  var wrap = document.createElement('div');
  wrap.id = 'tz-nav-wrap';
  wrap.setAttribute('role', 'group');
  wrap.setAttribute('aria-label', 'Taille des textes');
  wrap.innerHTML =
    '<button type="button" class="tz-btn" id="tz-down" aria-label="Diminuer la taille des textes" title="Diminuer la taille des textes">' +
      '<span class="tz-a" aria-hidden="true">A</span><i class="fas fa-minus" aria-hidden="true"></i>' +
    '</button>' +
    '<span id="tz-val" aria-live="polite">100 %</span>' +
    '<button type="button" class="tz-btn" id="tz-up" aria-label="Agrandir la taille des textes" title="Agrandir la taille des textes">' +
      '<span class="tz-a" aria-hidden="true">A</span><i class="fas fa-plus" aria-hidden="true"></i>' +
    '</button>';

  var themeBtn = document.getElementById('themeToggle');
  var navContent = document.getElementById('navContent');
  if (themeBtn && themeBtn.parentNode) themeBtn.parentNode.insertBefore(wrap, themeBtn);
  else if (navContent) navContent.appendChild(wrap);
  else return;

  var btnDown = document.getElementById('tz-down');
  var btnUp   = document.getElementById('tz-up');
  var out     = document.getElementById('tz-val');

  /* ── 2. Appliquer une taille ── */
  function apply(v, save) {
    scale = clamp(v);
    var pct = Math.round(scale * 100) + ' %';
    html.style.setProperty('--tz-scale', scale);
    html.classList.toggle('tz-on', scale !== 1);
    out.textContent = pct;
    btnDown.disabled = scale <= MIN;
    btnUp.disabled   = scale >= MAX;
    btnDown.title = 'Diminuer la taille des textes (actuellement ' + pct + ')';
    btnUp.title   = 'Agrandir la taille des textes (actuellement ' + pct + ')';
    if (save) {
      try { scale === 1 ? localStorage.removeItem(KEY) : localStorage.setItem(KEY, String(scale)); } catch (e) {}
    }
    fitNav();
  }

  btnDown.addEventListener('click', function () { apply(scale - STEP, true); });
  btnUp.addEventListener('click',   function () { apply(scale + STEP, true); });

  /* ── 3. La navbar ne déborde jamais ── */
  var nav = document.getElementById('mainNav');
  var bar = nav ? nav.querySelector('.container-fluid') : null;

  function overflows() {
    return navContent.scrollWidth > navContent.clientWidth + 1 || (bar && bar.scrollWidth > bar.clientWidth + 1);
  }

  function fitNav() {
    if (!nav || !navContent) return;
    /* On repart de l'affichage normal pour mesurer… */
    nav.classList.remove('tz-compact', 'tz-collapse');
    /* …sous 992 px, c'est déjà le menu burger de Bootstrap : rien à faire. */
    if (window.innerWidth < 992) return;
    /* Si la barre déborde : d'abord les icônes seules, puis, si besoin, le menu burger. */
    if (overflows()) {
      nav.classList.add('tz-compact');
      if (overflows()) nav.classList.add('tz-collapse');
    }
  }

  /* On remesure quand la fenêtre change de taille… */
  var raf = 0;
  function scheduleFit() { cancelAnimationFrame(raf); raf = requestAnimationFrame(fitNav); }
  window.addEventListener('resize', scheduleFit);
  window.addEventListener('orientationchange', scheduleFit);
  window.addEventListener('load', scheduleFit);
  if (document.fonts && document.fonts.ready) document.fonts.ready.then(scheduleFit);
  /* …et quand un autre module ajoute un bouton ou change un libellé (traduction, mode édition, Formspree…). */
  var ro = ('ResizeObserver' in window) ? new ResizeObserver(scheduleFit) : null;
  function watchChildren() { if (ro) Array.prototype.forEach.call(navContent.children, function (el) { ro.observe(el); }); }
  watchChildren();
  if ('MutationObserver' in window) {
    new MutationObserver(function () { watchChildren(); scheduleFit(); })
      .observe(navContent, { childList: true, subtree: true, characterData: true });
  }

  /* En mode burger forcé, un clic sur un lien referme le menu, comme sur mobile. */
  navContent.addEventListener('click', function (e) {
    if (!nav.classList.contains('tz-collapse') || !e.target.closest('.nav-link')) return;
    if (navContent.classList.contains('show') && window.bootstrap && bootstrap.Collapse) {
      bootstrap.Collapse.getOrCreateInstance(navContent, { toggle: false }).hide();
    }
  });

  /* ── 4. Au chargement : on reprend la taille mémorisée ── */
  var saved = 1;
  try { saved = parseFloat(localStorage.getItem(KEY)) || 1; } catch (e) {}
  apply(saved, false);
  /* Les autres modules insèrent leurs boutons après nous : on remesure un peu plus tard. */
  setTimeout(fitNav, 800);
  setTimeout(fitNav, 2500);
})();
</script>

J’ai ajouté ce bloc à une copie de votre page et je l’ai testé de 280 px à 2560 px de large, à 100 %, 115 % et 135 %, 
en mode clair et en mode sombre. La barre ne dépasse de l’écran dans aucun cas, et la taille choisie est bien retrouvée quand on recharge la page.

Pourquoi la barre s’adapte toute seule : 
votre barre actuelle déborde d’environ 200 px quand les textes sont à 135 %, à 992 px comme à 1920 px. 
Le code vérifie donc la place disponible. 
Si elle manque, les boutons n’affichent plus que leur icône. Si cela ne suffit pas, la barre passe en menu burger, comme sur mobile. 
À 100 %, la barre reste exactement comme aujourd’hui.

Pas de réduction sous 100 % : le bouton A− ramène vers la taille d’origine mais ne descend pas en dessous, 
pour ne pas passer sous la taille de texte minimale prévue dans votre feuille de style. 
Pour permettre une taille plus petite, il suffit de remplacer MIN = 1 par MIN = 0.9.
